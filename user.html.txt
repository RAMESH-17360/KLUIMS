    <script src="app.js"></script>
    <script>
        // Restrict page entry if not logged in
        const activeSession = JSON.parse(localStorage.getItem('currentUser'));
        if (!activeSession) {
            window.location.href = 'index.html';
        } else {
            document.getElementById('welcome-text').innerText = `Welcome, Operator Terminal Console`;
        }

        // Render User Inventory with Search Filtering Capability
        function renderUserInventory() {
            const dataList = JSON.parse(localStorage.getItem('inventory')) || [];
            const searchQuery = document.getElementById('search-input').value.toLowerCase();
            const container = document.getElementById('inventory-rows');
            container.innerHTML = '';
            
            let matchedItems = 0;

            dataList.forEach((product) => {
                // Filter matches by name
                if (!product.name.toLowerCase().includes(searchQuery)) return;
                
                matchedItems++;
                const isUnderThreshold = Number(product.stock) < Number(product.minStock);

                const displayRow = document.createElement('tr');
                if (isUnderThreshold) displayRow.classList.add('low-stock-row');

                displayRow.innerHTML = `
                    <td><code>${product.id}</code></td>
                    <td><strong>${product.name}</strong></td>
                    <td>${product.stock} units</td>
                    <td>${product.supplier}</td>
                    <td>
                        ${isUnderThreshold ? '<span class="badge-low">⚠️ REORDER NEEDED</span>' : '<span class="badge-ok">✓ AVAILABLE</span>'}
                    </td>
                `;
                container.appendChild(displayRow);
            });

            document.getElementById('total-skus').innerText = matchedItems;
        }

        // Run rendering sequence on document ready execution path
        window.onload = renderUserInventory;
    </script>
